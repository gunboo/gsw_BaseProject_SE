cmake_minimum_required(VERSION 3.24)
project(GSWGame VERSION 1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
option(GAME_STATIC_RUNTIME "Force a static MSVC release runtime (Debug is always static)" OFF)
if(MSVC)
  if(GAME_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    # Preserve the original release allocator/CRT performance; Debug needs no redistributable DLLs.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<IF:$<CONFIG:Debug>,Debug,DLL>")
  endif()
endif()

# These dependencies are versioned source trees, never network downloads at build time.
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build static dependencies" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
if(UNIX AND NOT APPLE)
  option(GLFW_BUILD_WAYLAND "Build the optional Wayland backend" OFF)
endif()
add_subdirectory(ThirdParty/glfw EXCLUDE_FROM_ALL)
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)
add_library(glew_static STATIC ThirdParty/glew/src/glew.c)
target_include_directories(glew_static SYSTEM PUBLIC ThirdParty/glew/include)
target_compile_definitions(glew_static PUBLIC GLEW_STATIC GLEW_NO_GLU)
target_link_libraries(glew_static PUBLIC OpenGL::GL)

file(GLOB GAME_SOURCES CONFIGURE_DEPENDS "SimpleGame/*.cpp")
list(REMOVE_ITEM GAME_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/SimpleGame/SimpleGame.cpp")
include(CheckIPOSupported)
check_ipo_supported(RESULT GAME_IPO_SUPPORTED LANGUAGES CXX)
if(GAME_IPO_SUPPORTED)
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
endif()
add_library(game_core STATIC ${GAME_SOURCES})
target_include_directories(game_core PUBLIC SimpleGame)
target_include_directories(game_core SYSTEM PRIVATE ThirdParty/stb)
target_link_libraries(game_core PUBLIC glew_static glfw Threads::Threads)
target_compile_definitions(game_core PUBLIC GLFW_INCLUDE_NONE)
if(WIN32)
  target_compile_definitions(game_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(game_core PUBLIC gdi32 psapi winhttp)
endif()
if(MSVC)
  target_compile_options(game_core PRIVATE /W3 /utf-8 /permissive-)
endif()
if(APPLE)
  target_compile_definitions(glew_static PRIVATE GL_SILENCE_DEPRECATION)
endif()

add_executable(SimpleGame SimpleGame/SimpleGame.cpp)
target_link_libraries(SimpleGame PRIVATE game_core)
if(WIN32)
  option(GAME_WINDOWS_NATIVE_DPI "Use native DPI instead of the original Windows display scaling" OFF)
  if(NOT GAME_WINDOWS_NATIVE_DPI)
    if(MSVC)
      target_sources(SimpleGame PRIVATE SimpleGame/Windows.manifest)
    else()
      enable_language(RC)
      target_sources(SimpleGame PRIVATE SimpleGame/Windows.rc)
    endif()
  endif()
endif()
set_property(TARGET SimpleGame PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "$<TARGET_FILE_DIR:SimpleGame>")
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT SimpleGame)
add_custom_target(game_assets ALL
  COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}/SimpleGame"
    "-DDESTINATION=${CMAKE_BINARY_DIR}/bin/$<CONFIG>" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CopyAssets.cmake"
  COMMENT "Copying authored data, shaders and licensed fonts")
add_dependencies(SimpleGame game_assets)
if(MSVC AND NOT GAME_STATIC_RUNTIME)
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
  include(InstallRequiredSystemLibraries)
  if(NOT CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS)
    message(FATAL_ERROR "MSVC redistributable DLLs were not found. Install the matching Visual Studio C++ tools, or use GAME_STATIC_RUNTIME=ON.")
  endif()
  add_custom_target(game_runtime ALL
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS}
      "${CMAKE_BINARY_DIR}/bin/$<CONFIG>/"
    COMMENT "Copying the matching app-local MSVC release runtime")
  add_dependencies(game_runtime game_assets)
  add_dependencies(SimpleGame game_runtime)
  install(PROGRAMS ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION .
    CONFIGURATIONS Release RelWithDebInfo MinSizeRel)
endif()

include(CTest)
if(BUILD_TESTING)
  foreach(name IN ITEMS player_motion resident_activity render_optimizations fonts window_input map)
    add_executable(test_${name} Tools/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE game_core)
    add_dependencies(test_${name} game_assets)
    if(TARGET game_runtime)
      add_dependencies(test_${name} game_runtime)
    endif()
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>" TIMEOUT 120)
  endforeach()
  option(GAME_GRAPHICS_TESTS "Run native window/OpenGL smoke tests (requires a display)" OFF)
  if(GAME_GRAPHICS_TESTS)
    foreach(scene IN ITEMS tutorial level1)
      add_test(NAME graphics_${scene} COMMAND SimpleGame)
      set_tests_properties(graphics_${scene} PROPERTIES TIMEOUT 120 LABELS graphics
        ENVIRONMENT "GAME_SMOKE_SECONDS=2;GAME_SMOKE_SCENE=${scene};GAME_MONITOR_ENABLED=0")
    endforeach()
  endif()
endif()

install(TARGETS SimpleGame RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES docs/GLFW_MIGRATION.md DESTINATION docs)
install(FILES ThirdParty/README.md ThirdParty/manifest.json DESTINATION ThirdParty)
install(FILES ThirdParty/glfw/LICENSE.md DESTINATION ThirdParty/glfw)
install(FILES ThirdParty/glew/LICENSE.txt DESTINATION ThirdParty/glew)
install(FILES ThirdParty/stb/LICENSE DESTINATION ThirdParty/stb)
install(DIRECTORY SimpleGame/Data SimpleGame/Shaders DESTINATION .
  PATTERN "Logs" EXCLUDE PATTERN "*.cache" EXCLUDE PATTERN ".gitignore" EXCLUDE)
install(FILES ThirdParty/glfw/LICENSE.md DESTINATION licenses RENAME GLFW-LICENSE.md)
install(FILES ThirdParty/glew/LICENSE.txt DESTINATION licenses RENAME GLEW-LICENSE.txt)
install(FILES ThirdParty/stb/LICENSE DESTINATION licenses RENAME STB-LICENSE.txt)
set(CPACK_GENERATOR ZIP)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(GAME_PACKAGE_ARCH x86)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]|aarch64")
  set(GAME_PACKAGE_ARCH arm64)
else()
  set(GAME_PACKAGE_ARCH x64)
endif()
set(CPACK_PACKAGE_FILE_NAME "GSWGame-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${GAME_PACKAGE_ARCH}")
include(CPack)
